    let errores = 0;
    const maxIntentos = 2;
    let musicaIniciada = false;
    let lluviaActiva = false;

    function verificarRespuestas() {
      const q1 = document.querySelector('input[name="q1"]:checked');
      const q2 = document.querySelector('input[name="q2"]:checked');
      const result = document.getElementById('result');
      const locked = document.getElementById('locked');
      const surprise = document.getElementById('surprise');
      const quiz = document.getElementById('quiz');
      const blockedPage = document.getElementById('blockedPage');
      const attemptsBox = document.getElementById('attemptsBox');
      const unlockBtn = document.getElementById('unlockBtn');

      if (!q1 || !q2) {
        result.textContent = 'Tenés que responder las dos preguntas primero 💌';
        return;
      }

      if (q1.value === 'v' && q2.value === 'f') {
        result.textContent = 'Correcto, mensaje desbloqueado ❤️';
        locked.textContent = '🔓';
        attemptsBox.textContent = '¡Correcto! Sorpresa desbloqueada ❤️';

        setTimeout(() => {
          quiz.style.display = 'none';
          surprise.classList.add('show');
          iniciarMusicaRomantica();
          activarCorazones();
        }, 800);
      } else {
        errores++;
        const intentosRestantes = maxIntentos - errores;

        if (intentosRestantes > 0) {
          result.textContent = 'Casi, pero hay una respuesta incorrecta. Te queda 1 intento 😺';
          attemptsBox.textContent = 'Solo tienes 2 intentos · Intentos restantes: ' + intentosRestantes;
          crearCorazones(8);
        } else {
          result.textContent = 'Se acabaron los intentos.';
          attemptsBox.textContent = 'Solo tienes 2 intentos · Intentos restantes: 0';
          locked.textContent = '🔒';
          unlockBtn.disabled = true;

          document.querySelectorAll('input[type="radio"]').forEach((input) => {
            input.disabled = true;
          });

          setTimeout(() => {
            quiz.style.display = 'none';
            blockedPage.classList.add('show');
            crearCorazones(15);
          }, 800);
        }
      }
    }

    function activarCorazones() {
      if (lluviaActiva) return;
      lluviaActiva = true;
      crearCorazones(35);
      setInterval(() => crearCorazon(), 450);
    }

    function crearCorazones(cantidad) {
      for (let i = 0; i < cantidad; i++) {
        setTimeout(crearCorazon, i * 80);
      }
    }

    function crearCorazon() {
      const corazon = document.createElement('div');
      corazon.classList.add('heart');
      corazon.innerHTML = Math.random() > 0.5 ? '❤️' : '💕';
      corazon.style.left = Math.random() * 100 + 'vw';
      corazon.style.fontSize = (18 + Math.random() * 34) + 'px';
      corazon.style.animationDuration = (3 + Math.random() * 3) + 's';
      document.body.appendChild(corazon);

      setTimeout(() => {
        corazon.remove();
      }, 6500);
    }

    function iniciarMusicaRomantica() {
      if (musicaIniciada) return;
      musicaIniciada = true;

      const AudioContext = window.AudioContext || window.webkitAudioContext;
      const audioCtx = new AudioContext();

      const master = audioCtx.createGain();
      master.gain.value = 0.12;
      master.connect(audioCtx.destination);

      const notas = [
        { f: 392.00, t: 0.0 },
        { f: 440.00, t: 0.45 },
        { f: 493.88, t: 0.9 },
        { f: 587.33, t: 1.35 },
        { f: 523.25, t: 1.8 },
        { f: 493.88, t: 2.25 },
        { f: 440.00, t: 2.7 },
        { f: 392.00, t: 3.15 }
      ];

      function tocarMelodia() {
        const ahora = audioCtx.currentTime;

        notas.forEach((nota) => {
          const osc = audioCtx.createOscillator();
          const gain = audioCtx.createGain();

          osc.type = 'sine';
          osc.frequency.setValueAtTime(nota.f, ahora + nota.t);

          gain.gain.setValueAtTime(0, ahora + nota.t);
          gain.gain.linearRampToValueAtTime(0.45, ahora + nota.t + 0.05);
          gain.gain.exponentialRampToValueAtTime(0.001, ahora + nota.t + 0.42);

          osc.connect(gain);
          gain.connect(master);

          osc.start(ahora + nota.t);
          osc.stop(ahora + nota.t + 0.5);
        });
      }

      tocarMelodia();
      setInterval(tocarMelodia, 3900);
    }
